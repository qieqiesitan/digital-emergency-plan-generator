### 次要（顺手）：localStorage 防护

`getItem`/`setItem` 包 try/catch（隐私模式/存储禁用时不白屏）：

```tsx
const getStoredProMode = () => {
  try { return localStorage.getItem("pro_mode") === "1"; } catch { return false; }
};
const setStoredProMode = (v: boolean) => {
  try { localStorage.setItem("pro_mode", v ? "1" : "0"); } catch { /* 存储不可用时忽略 */ }
};
```
