**service**（箭头函数 + `.then(r=>r.data.data)` 解包，与项目惯例一致）：

```typescript
export const getOrgNodes = (enterpriseId: string) => api.get(`/enterprises/${enterpriseId}/org/nodes`).then(r => r.data.data);
export const saveOrgNodes = (enterpriseId: string, nodes: OrgNode[]) => api.put(`/enterprises/${enterpriseId}/org/nodes`, { nodes }).then(r => r.data.data);
export const listMembers = (enterpriseId: string) => api.get(`/enterprises/${enterpriseId}/org/members`).then(r => r.data.data);
export const createMember = (enterpriseId: string, payload: object) => api.post(`/enterprises/${enterpriseId}/org/members`, payload).then(r => r.data.data);
export const updateMember = (enterpriseId: string, memberId: string, patch: object) => api.put(`/enterprises/${enterpriseId}/org/members/${memberId}`, patch).then(r => r.data.data);
export const deleteMember = (enterpriseId: string, memberId: string) => api.delete(`/enterprises/${enterpriseId}/org/members/${memberId}`).then(r => r.data.data);
export const importMembers = (enterpriseId: string, file: File) => { const fd = new FormData(); fd.append("file", file); return api.post(`/enterprises/${enterpriseId}/org/members/import`, fd).then(r => r.data.data); };
export const suggestOrgTree = (enterpriseId: string) => api.post(`/enterprises/${enterpriseId}/org/ai-suggest`).then(r => r.data.data);
```

（后端实际端点路径以前端为准：注意任务是 `/org/nodes`、`/org/members`、`/org/members/import`、`/org/ai-suggest`——若与实现路径有出入，以后端实际为准并同步；下载模板用 `api.get('/enterprises/{id}/org/members/template', {responseType:'blob'})`——若后端无此端点，用 `build_member_import_template` 对应的下载端点；先读后端 router 确认所有端点路径与下载能力，缺下载端点则后端补一个 GET template 端点并提交。）
